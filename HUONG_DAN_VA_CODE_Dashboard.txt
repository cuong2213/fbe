Public Sub DeleteLegacyChart(ByVal ws As Worksheet, ByVal nm As String)
    On Error Resume Next
    ws.ChartObjects(nm).Delete
    On Error GoTo 0
End Sub

Private Function EnsurePieChart(ByVal ws As Worksheet) As ChartObject
    Dim co As ChartObject, rng As Range
    Set rng = ws.Range("H45:N58")
    On Error Resume Next
    Set co = ws.ChartObjects("chtDefectPie")
    On Error GoTo 0
    If co Is Nothing Then
        Set co = ws.ChartObjects.Add(rng.Left + 2, rng.Top + 2, rng.Width - 4, rng.Height - 4)
        co.Name = "chtDefectPie"
        With co.Chart
            .ChartType = xlPie
            .HasTitle = True
            .ChartTitle.Text = "DEFECT RATIO BY DEFECT TYPE (%)"
            .ChartTitle.Font.Size = 11
            .ChartTitle.Font.Bold = True
        End With
        On Error Resume Next
        co.Chart.ChartArea.Format.Fill.ForeColor.RGB = RGB(255, 255, 255)
        co.Chart.ChartArea.Format.Line.ForeColor.RGB = RGB(200, 200, 200)
        co.Chart.ChartArea.Font.Size = 9
        On Error GoTo 0
    Else
        co.Left = rng.Left + 2: co.Top = rng.Top + 2
        co.Width = rng.Width - 4: co.Height = rng.Height - 4
    End If
    co.Placement = xlMoveAndSize
    Set EnsurePieChart = co
End Function

Public Sub CreatePieChart(ByVal ws As Worksheet)
    Dim co As ChartObject
    Set co = EnsurePieChart(ws)
End Sub

Public Sub UpdatePieChart()
    Dim ws As Worksheet, calc As Worksheet, co As ChartObject, cht As Chart, s As Series
    Dim ks() As String, vs() As Double, n As Long, m As Long, i As Long, nOut As Long
    Dim others As Double, arr() As Variant, clr As Long
    Const C0 As Long = 40
    Set ws = DashSheet(SH_DASH)
    Set calc = DashSheet(SH_CALC)
    DeleteLegacyChart ws, "chtTop3"
    Set co = EnsurePieChart(ws)
    Set cht = co.Chart
    ClearSeries cht
    calc.Range(calc.Cells(1, C0), calc.Cells(40, C0 + 1)).ClearContents

    n = SortDictKeys(rDefQty, ks, vs, True)
    If n = 0 Then
        co.Visible = False
        Exit Sub
    End If
    co.Visible = True

    If n > PIE_MAX_SLICES Then m = PIE_MAX_SLICES - 1 Else m = n
    For i = m + 1 To n
        others = others + vs(i)
    Next i
    nOut = m
    If others > 0 Then nOut = m + 1
    ReDim arr(1 To nOut + 1, 1 To 2)
    arr(1, 1) = "Defect Type": arr(1, 2) = "Defects"
    For i = 1 To m
        arr(i + 1, 1) = ks(i): arr(i + 1, 2) = vs(i)
    Next i
    If others > 0 Then
        arr(nOut + 1, 1) = "Others": arr(nOut + 1, 2) = others
    End If
    calc.Cells(1, C0).Resize(nOut + 1, 2).Value = arr

    cht.ChartType = xlPie
    Set s = cht.SeriesCollection.NewSeries
    s.Name = "Defects"
    s.XValues = calc.Range(calc.Cells(2, C0), calc.Cells(nOut + 1, C0))
    s.Values = calc.Range(calc.Cells(2, C0 + 1), calc.Cells(nOut + 1, C0 + 1))
    s.HasDataLabels = True
    With s.DataLabels
        .ShowPercentage = True
        .ShowValue = False
        .ShowCategoryName = False
        .NumberFormat = "0.0%"
        .Position = xlLabelPositionOutsideEnd
        .Font.Size = 9
    End With
    For i = 1 To nOut
        If i <= m Then clr = DefectColor(ks(i)) Else clr = RGB(166, 166, 166)
        s.Points(i).Format.Fill.ForeColor.RGB = clr
        If rTotal > 0 Then
            s.Points(i).DataLabel.Text = CStr(Round(arr(i + 1, 2) / rTotal * 100, 1)) & "% (" & CStr(Round(arr(i + 1, 2), 2)) & "/" & CStr(Round(rTotal, 2)) & ")"
        End If
    Next i
    cht.HasLegend = True
    cht.Legend.Position = xlLegendPositionRight
End Sub

' VEHICLE DEFECT HEATMAP - Top 3 vi tri loi (o Grid) cho tung Defect Type, tung Model
' - Du lieu: rHeat (Model|DefectType|hang,cot -> so loi) do LoadAndAggregate tong hop theo bo loc Dashboard
' - Hinh xe: hinh Master (tblModels > ImageShapeName) tren MAP_VEHICLE; Grid chia deu theo Rows x Cols cua Model,
'   nen o (r, c) nam tai ((c - 0.5) / Cols, (r - 0.5) / Rows) cua hinh
' - Bang Top 3 cu da bo; hang 60-75 dung cho bieu do ti le loi / xe
'=====================================================================
Public Sub UpdateHeatmap(ByVal showMsg As Boolean)
    Dim ws As Worksheet, mdl As Object, grp As Object, nSkip As Long
    Dim defs() As String, vsD() As Double, nAll As Long, nD As Long
    Dim mods() As String, nM As Long, k As Variant
    Dim availW As Double, availH As Double, legendH As Double, hmAvail As Double
    Dim pics() As Shape, asp() As Double, perRow As Long, nTileRows As Long
    Dim i As Long, tr As Long, tc As Long, tileW As Double, sumAsp As Double, mx As Double
    Dim yRow() As Double, x As Double, y As Double
    Dim info As Variant, s As Shape, pic As Shape, dd As Long, rr As Long
    Dim t As Variant, key As String, occ As Object, n As Long, base As Double, sz As Double
    Dim mx0 As Double, my0 As Double, rowN As Long, colN As Long, v As Double, clr As Long, tip As String

    On Error GoTo EH
    Set ws = DashSheet(SH_DASH)

    availW = ws.Range("O1").Left - ws.Range("A1").Left - 8
    availH = ws.Rows(ROW_DT_HDR - 1).Top - ws.Rows(HM_FIRST_ROW).Top - 6
    If availH < 120 Then
        If showMsg Then MsgBox "Vung Vehicle Heatmap qua thap (hang " & HM_FIRST_ROW & " den " & (ROW_DT_HDR - 2) & "). Hay tang ROW_DT_HDR.", vbExclamation, "Dashboard"
        GoTo CLEANUP
    End If

    HmClearShapes ws
    ws.Range("A93").Value = "So 1-2-3 = thu hang vi tri loi (o Grid) cua tung Defect Type, theo tung Model. Mau = Defect Type (Config). Re chuot len so de xem Cell ID."

    If rHeat Is Nothing Then GoTo CLEANUP
    HmDeleteStalePics ws, rModelQty
    If rHeat.Count = 0 Or rModelQty.Count = 0 Then
        ws.Range("A93").Value = "Khong co du lieu phu hop bo loc hien tai."
        GoTo CLEANUP
    End If

    Set mdl = HmReadModels()
    HmBuildTop3 mdl, grp, nSkip

    'Defect Type theo tong giam dan (toi da HM_MAX_TYPES)
    nAll = SortDictKeys(rDefQty, defs, vsD, True)
    nD = nAll
    If nD > HM_MAX_TYPES Then nD = HM_MAX_TYPES
    If nD = 0 Then GoTo CLEANUP

    'Danh sach Model co du lieu, theo thu tu tblModels
    ReDim mods(1 To rModelQty.Count)
    For Each k In mdl.Keys
        If rModelQty.Exists(CStr(k)) Then
            nM = nM + 1
            mods(nM) = CStr(k)
        End If
    Next k
    If nM = 0 Then
        ws.Range("A93").Value = "Model trong du lieu khong co trong MASTER > tblModels nen khong ve duoc heatmap."
        GoTo CLEANUP
    End If

    'Hinh nen cho tung Model
    ReDim pics(1 To nM)
    ReDim asp(1 To nM)
    For i = 1 To nM
        info = mdl(mods(i))
        Set pics(i) = HmModelPicture(ws, mods(i), CStr(info(2)), showMsg)
        If pics(i) Is Nothing Then
            If showMsg Then MsgBox "Khong lay duoc hinh Master '" & CStr(info(2)) & "' cua Model " & mods(i) & ".", vbExclamation, "Dashboard"
            asp(i) = 0.5
        Else
            asp(i) = pics(i).Height / pics(i).Width
        End If
    Next i

    'Bo cuc: legend, roi cac o (tile) 1 hoac 2 cot
    legendH = HmDrawLegend(ws, defs, nD, availW)
    hmAvail = availH - legendH - 4
    If nM = 1 Then perRow = 1 Else perRow = 2
    nTileRows = (nM + perRow - 1) \ perRow
    tileW = (availW - HM_GAP * (perRow - 1)) / perRow
    ReDim yRow(0 To nTileRows - 1)
    For tr = 0 To nTileRows - 1
        mx = 0
        For tc = 0 To perRow - 1
            i = tr * perRow + tc + 1
            If i <= nM Then
                If asp(i) > mx Then mx = asp(i)
            End If
        Next tc
        sumAsp = sumAsp + mx
    Next tr
    If tileW * sumAsp + nTileRows * HM_TITLE_H + HM_GAP * (nTileRows - 1) > hmAvail Then
        tileW = (hmAvail - nTileRows * HM_TITLE_H - HM_GAP * (nTileRows - 1)) / sumAsp
    End If
    If tileW < 120 Then
        If showMsg Then MsgBox "Vung Vehicle Heatmap qua thap cho " & nM & " Model. Hay tang ROW_DT_HDR.", vbExclamation, "Dashboard"
        GoTo CLEANUP
    End If

    y = ws.Rows(HM_FIRST_ROW).Top + 2 + legendH + 4
    For tr = 0 To nTileRows - 1
        yRow(tr) = y
        mx = 0
        For tc = 0 To perRow - 1
            i = tr * perRow + tc + 1
            If i <= nM Then
                If asp(i) > mx Then mx = asp(i)
            End If
        Next tc
        y = y + HM_TITLE_H + tileW * mx + HM_GAP
    Next tr

    base = tileW / 45
    If base < 11 Then base = 11
    If base > 20 Then base = 20
    Set occ = CreateObject("Scripting.Dictionary")

    For i = 1 To nM
        tr = (i - 1) \ perRow
        tc = (i - 1) Mod perRow
        x = ws.Range("A1").Left + 2 + tc * (tileW + HM_GAP)
        info = mdl(mods(i))

        'tieu de o
        Set s = ws.Shapes.AddShape(msoShapeRectangle, x, yRow(tr), tileW, HM_TITLE_H)
        With s
            .Name = "HM_T_" & mods(i)
            .Fill.ForeColor.RGB = RGB(31, 56, 100)
            .Line.Visible = msoFalse
            .Placement = xlMove
            With .TextFrame2
                .MarginLeft = 4: .MarginRight = 4: .MarginTop = 0: .MarginBottom = 0
                .VerticalAnchor = msoAnchorMiddle
                .WordWrap = msoFalse
                .TextRange.Text = mods(i) & "  (" & info(0) & " x " & info(1) & ")  -  " & Format$(CDbl(rModelQty(mods(i))), "#,##0")
                .TextRange.Font.Size = 9
                .TextRange.Font.Bold = msoTrue
                .TextRange.Font.Fill.ForeColor.RGB = RGB(255, 255, 255)
                .TextRange.ParagraphFormat.Alignment = msoAlignLeft
            End With
        End With

        'hinh nen
        Set pic = pics(i)
        If Not pic Is Nothing Then
            With pic
                .LockAspectRatio = msoFalse
                .Left = x
                .Top = yRow(tr) + HM_TITLE_H
                .Width = tileW
                .Height = tileW * asp(i)
                .Line.Visible = msoTrue
                .Line.ForeColor.RGB = RGB(150, 150, 150)
                .Line.Weight = 0.75
            End With
            mx0 = pic.Left: my0 = pic.Top

            'marker Top 3 cho tung Defect Type
            For dd = 1 To nD
                key = mods(i) & "|" & defs(dd)
                If grp.Exists(key) Then
                    t = grp(key)
                    clr = DefectColor(defs(dd))
                    For rr = 1 To 3
                        v = t(rr, 1)
                        If v > 0 Then
                            rowN = CLng(t(rr, 2)): colN = CLng(t(rr, 3))
                            sz = base * (1 - 0.15 * (rr - 1))
                            n = 0
                            If occ.Exists(i & "|" & rowN & "," & colN) Then n = occ(i & "|" & rowN & "," & colN)
                            occ(i & "|" & rowN & "," & colN) = n + 1
                            Set s = ws.Shapes.AddShape(msoShapeOval, _
                                    mx0 + (colN - 0.5) / CLng(info(1)) * pic.Width + n * sz * 0.45 - sz / 2, _
                                    my0 + (rowN - 0.5) / CLng(info(0)) * pic.Height - sz / 2, sz, sz)
                            tip = mods(i) & " - " & defs(dd) & " - hang " & rr & " - " & ColumnLetter(colN) & rowN & " (" & CStr(CLng(v)) & ")"
                            HmStyleMarker s, "HM_M_" & i & "_" & dd & "_" & rr, CStr(rr), clr, tip, sz
                        End If
                    Next rr
                End If
            Next dd
        End If
    Next i

    If nSkip > 0 Then
        ws.Range("A93").Value = ws.Range("A93").Value & "  (Bo qua " & nSkip & " ban ghi ngoai Grid)"
    End If
CLEANUP:
    Application.CutCopyMode = False
    Exit Sub
EH:
    If showMsg Then MsgBox "Heatmap loi: " & Err.Description, vbExclamation, "Dashboard"
    Resume CLEANUP
End Sub

'Model -> Array(Rows, Cols, ImageShapeName), thu tu theo MASTER > tblModels
Private Function HmReadModels() As Object
    Dim d As Object, lo As ListObject, dat As Variant, i As Long, nm As String
    Dim cM As Long, cR As Long, cC As Long, cI As Long
    Set d = NewDict()
    Set lo = DashTable(SH_MASTER, TBL_MODELS)
    If Not lo.DataBodyRange Is Nothing Then
        cM = DashColIndex(lo, "Model")
        cR = DashColIndex(lo, "Rows")
        cC = DashColIndex(lo, "Cols")
        cI = DashColIndex(lo, "ImageShapeName")
        dat = lo.DataBodyRange.Value2
        For i = 1 To UBound(dat, 1)
            nm = Trim$(CStr(dat(i, cM)))
            If Len(nm) > 0 And IsNumeric(dat(i, cR)) And IsNumeric(dat(i, cC)) Then
                If CLng(dat(i, cR)) >= 1 And CLng(dat(i, cC)) >= 1 And Not d.Exists(nm) Then
                    d.Add nm, Array(CLng(dat(i, cR)), CLng(dat(i, cC)), Trim$(CStr(dat(i, cI))))
                End If
            End If
        Next i
    End If
    Set HmReadModels = d
End Function

'grp("Model|DefectType") = mang 3x3: (hang thu, 1 = so loi / 2 = hang Grid / 3 = cot Grid)
Private Sub HmBuildTop3(ByVal mdl As Object, ByRef grp As Object, ByRef nSkip As Long)
    Dim k As Variant, p() As String, rc() As String, gk As String
    Dim r As Long, c As Long, v As Double, info As Variant
    Dim t As Variant, blank(1 To 3, 1 To 3) As Double
    Set grp = NewDict()
    nSkip = 0
    For Each k In rHeat.Keys
        p = Split(CStr(k), "|")
        If UBound(p) = 2 Then
            If mdl.Exists(p(0)) Then
                rc = Split(p(2), ",")
                r = CLng(rc(0)): c = CLng(rc(1))
                info = mdl(p(0))
                If r < 1 Or r > CLng(info(0)) Or c < 1 Or c > CLng(info(1)) Then
                    nSkip = nSkip + 1
                Else
                    v = CDbl(rHeat(k))
                    gk = p(0) & "|" & p(1)
                    If grp.Exists(gk) Then
                        t = grp(gk)
                    Else
                        Erase blank
                        t = blank
                    End If
                    HmInsertTop3 t, v, r, c
                    grp(gk) = t
                End If
            End If
        End If
    Next k
End Sub

Private Sub HmInsertTop3(ByRef t As Variant, ByVal v As Double, ByVal r As Long, ByVal c As Long)
    Dim p As Long, q As Long
    For p = 1 To 3
        If v > t(p, 1) Then
            For q = 3 To p + 1 Step -1
                t(q, 1) = t(q - 1, 1): t(q, 2) = t(q - 1, 2): t(q, 3) = t(q - 1, 3)
            Next q
            t(p, 1) = v: t(p, 2) = r: t(p, 3) = c
            Exit Sub
        End If
    Next p
End Sub

Private Function HmFindShape(ByVal ws As Worksheet, ByVal nm As String) As Shape
    On Error Resume Next
    Set HmFindShape = ws.Shapes(nm)
    On Error GoTo 0
End Function

'Uu tien hinh that; hinh tam (AutoShape) chi dung khi chua co hinh that
Private Function HmFindImage(ByVal wm As Worksheet, ByVal nm As String) As Shape
    Dim s As Shape, temp As Shape
    For Each s In wm.Shapes
        If StrComp(s.Name, nm, vbTextCompare) = 0 Then
            If s.Type = msoAutoShape Then
                If temp Is Nothing Then Set temp = s
            Else
                Set HmFindImage = s
                Exit Function
            End If
        End If
    Next s
    Set HmFindImage = temp
End Function

'Copy hinh Master cua Model vao Dashboard (tam bat hien thi neu dang an, roi tra lai nhu cu).

Private Function HmModelPicture(ByVal ws As Worksheet, ByVal model As String, ByVal imgName As String, _
                                ByVal force As Boolean) As Shape
    Dim nm As String, s As Shape, src As Shape, wm As Worksheet, wasVis As Boolean
    Dim tries As Long, ok As Boolean
    nm = "HM_PIC_" & model
    Set s = HmFindShape(ws, nm)
    If Not s Is Nothing Then
        If Not force Then
            Set HmModelPicture = s
            Exit Function
        End If
        s.Delete
        Set s = Nothing
    End If
    If Len(imgName) = 0 Then Exit Function
    Set wm = DashSheet(SH_MAP)
    Set src = HmFindImage(wm, imgName)
    If src Is Nothing Then Exit Function

    wasVis = (src.Visible = msoTrue)
    If Not wasVis Then src.Visible = msoTrue
    For tries = 1 To 3
        On Error Resume Next
        Err.Clear
        src.Copy
        If Err.Number = 0 Then ws.Paste Destination:=ws.Range("A" & HM_FIRST_ROW)
        ok = (Err.Number = 0)
        On Error GoTo 0
        If ok Then Exit For
        DoEvents
    Next tries
    On Error Resume Next
    If Not wasVis Then src.Visible = msoFalse
    On Error GoTo 0
    If Not ok Then Exit Function

    Set s = ws.Shapes(ws.Shapes.Count)
    s.Name = nm
    s.Placement = xlMove
    Set HmModelPicture = s
End Function

Private Function HmDrawLegend(ByVal ws As Worksheet, ByRef defs() As String, ByVal nD As Long, _
                              ByVal availW As Double) As Double
    Dim i As Long, x As Double, y As Double, w As Double, x0 As Double, y0 As Double
    Dim clr As Long, s As Shape
    x0 = ws.Range("A1").Left + 2
    y0 = ws.Rows(HM_FIRST_ROW).Top + 2
    x = x0: y = y0
    For i = 1 To nD
        w = Len(defs(i)) * 5.6 + 18
        If w < 44 Then w = 44
        If x + w > x0 + availW And x > x0 Then
            x = x0
            y = y + 18
        End If
        clr = DefectColor(defs(i))
        Set s = ws.Shapes.AddShape(msoShapeRoundedRectangle, x, y, w, 15)
        With s
            .Name = "HM_L_" & i
            .Fill.ForeColor.RGB = clr
            .Line.Visible = msoFalse
            .Placement = xlMove
            With .TextFrame2
                .MarginLeft = 2: .MarginRight = 2: .MarginTop = 0: .MarginBottom = 0
                .VerticalAnchor = msoAnchorMiddle
                .WordWrap = msoFalse
                .TextRange.Text = defs(i)
                .TextRange.Font.Size = 8
                .TextRange.Font.Bold = msoTrue
                .TextRange.Font.Fill.ForeColor.RGB = HmTextColor(clr)
                .TextRange.ParagraphFormat.Alignment = msoAlignCenter
            End With
        End With
        x = x + w + 6
    Next i
    HmDrawLegend = (y - y0) + 15
End Function

Private Function HmTextColor(ByVal clr As Long) As Long
    Dim lum As Double
    lum = 0.299 * (clr And 255) + 0.587 * ((clr \ 256) And 255) + 0.114 * ((clr \ 65536) And 255)
    If lum > 150 Then HmTextColor = RGB(0, 0, 0) Else HmTextColor = RGB(255, 255, 255)
End Function

'Mau to cot = ma Hex trong MASTER > tblColors (khong co -> xam)
Private Function PaintRGB(ByVal colorName As String) As Long
    Dim lo As ListObject, i As Long, dat As Variant, h As String
    PaintRGB = RGB(166, 166, 166)
    On Error GoTo Done
    Set lo = DashTable(SH_MASTER, TBL_COLORS)
    If lo.DataBodyRange Is Nothing Then Exit Function
    dat = lo.DataBodyRange.Value2
    For i = 1 To UBound(dat, 1)
        If StrComp(Trim$(CStr(dat(i, 1))), colorName, vbTextCompare) = 0 Then
            h = Replace(Trim$(CStr(dat(i, 2))), "#", "")
            If Len(h) = 6 Then
                PaintRGB = RGB(CLng("&H" & Mid$(h, 1, 2)), CLng("&H" & Mid$(h, 3, 2)), CLng("&H" & Mid$(h, 5, 2)))
            End If
            Exit Function
        End If
    Next i
Done:
End Function

Public Sub DashCalOnSelection(ByVal Target As Range)
    Dim ws As Worksheet
    On Error Resume Next
    Set ws = Target.Worksheet
    If Target.Cells.Count > 2 Then
        DashCalHide ws
        Exit Sub
    End If
    If Not Intersect(Target, ws.Range(F_FROM).MergeArea) Is Nothing Then
        DashCalShow ws, F_FROM
    ElseIf Not Intersect(Target, ws.Range(F_TO).MergeArea) Is Nothing Then
        DashCalShow ws, F_TO
    Else
        DashCalHide ws
    End If
End Sub

End Sub

Private Sub DashCalLayout(ByVal ws As Worksheet, ByVal L As Single, ByVal T As Single)
    Dim i As Long, totalW As Single
    Const CW As Single = 26
    Const CH As Single = 20
    totalW = 7 * CW + 10
    With ws.Shapes("CAL_BG")
        .Left = L: .Top = T: .Width = totalW: .Height = 196
    End With
    With ws.Shapes("CAL_PREV")
        .Left = L + 5: .Top = T + 4: .Width = CW: .Height = 20
    End With
    With ws.Shapes("CAL_NEXT")
        .Left = L + 5 + 6 * CW: .Top = T + 4: .Width = CW: .Height = 20
    End With
    With ws.Shapes("CAL_TITLE")
        .Left = L + 5 + CW: .Top = T + 4: .Width = 5 * CW: .Height = 20
    End With
    For i = 1 To 7
        With ws.Shapes("CAL_W" & i)
            .Left = L + 5 + (i - 1) * CW: .Top = T + 26: .Width = CW: .Height = 16
        End With
    Next i
    For i = 1 To 42
        With ws.Shapes("CAL_D" & i)
            .Left = L + 5 + ((i - 1) Mod 7) * CW
            .Top = T + 44 + ((i - 1) \ 7) * CH
            .Width = CW: .Height = CH
        End With
    Next i
    With ws.Shapes("CAL_TODAY")
        .Left = L + 5: .Top = T + 44 + 6 * CH + 4: .Width = totalW - 10: .Height = 20
    End With
End Sub

Private Sub DashCalDraw(ByVal ws As Worksheet)
    Dim firstDay As Date, offset As Long, daysIn As Long, i As Long, d As Long
    Dim dd As Date, sel As Date, s As Shape, v As Variant
    firstDay = DateSerial(mDcYear, mDcMonth, 1)
    offset = Weekday(firstDay, vbMonday) - 1
    daysIn = Day(DateSerial(mDcYear, mDcMonth + 1, 0))
    ws.Shapes("CAL_TITLE").TextFrame2.TextRange.Text = "Thang " & mDcMonth & "/" & mDcYear
    v = ws.Range(mDcTarget).Value
    If IsDate(v) Then sel = CDate(v) Else sel = 0
    For i = 1 To 42
        d = i - offset
        Set s = ws.Shapes("CAL_D" & i)
        If d >= 1 And d <= daysIn Then
            dd = DateSerial(mDcYear, mDcMonth, d)
            s.TextFrame2.TextRange.Text = CStr(d)
            s.Fill.ForeColor.RGB = RGB(255, 255, 255)
            s.TextFrame2.TextRange.Font.Fill.ForeColor.RGB = IIf(((i - 1) Mod 7) = 6, RGB(192, 0, 0), RGB(0, 0, 0))
            If dd = Date Then s.Fill.ForeColor.RGB = RGB(255, 230, 153)
            If sel <> 0 And dd = sel Then
                s.Fill.ForeColor.RGB = RGB(31, 78, 121)
                s.TextFrame2.TextRange.Font.Fill.ForeColor.RGB = RGB(255, 255, 255)
            End If
            s.Visible = msoTrue
        Else
            s.Visible = msoFalse
        End If
    Next i
End Sub

        mDcMonth = 1
        mDcYear = mDcYear + 1

    DashCalDraw ActiveSheet
End Sub

Public Sub DashCalPick()

    If Len(mDcTarget) = 0 Then Exit Sub
    Set ws = ActiveSheet
    nm = CStr(Application.Caller)

    DashCalCommit ws, DateSerial(mDcYear, mDcMonth, d)
End Sub

Public Sub DashCalToday()
    If Len(mDcTarget) = 0 Then Exit Sub
    DashCalCommit ActiveSheet, Date
End Sub

    ws.Range("A7").Select                   'de lan sau bam lai vao o ngay van hien lich
End Sub

Public Sub EnsureColorSupport(ByVal ws As Worksheet)
    'Idempotent. Goi tu InitWorkbook (sheet MAP_VEHICLE dang UNPROTECT).
    Dim wsM As Worksheet, lo As ListObject
    On Error GoTo EH
    Set wsM = GetSheet(SH_MASTER)
    EnsureTable wsM, TBL_COLORS, "AC1", Array("PaintColor", "Hex"), _
        Array(Array("White", "#FFFFFF"), Array("Black", "#000000"), Array("Silver", "#C0C0C0"), _
              Array("Gray", "#808080"), Array("Red", "#C00000"), Array("Blue", "#1F4E79"), _
              Array("Green", "#2E7D32"), Array("Yellow", "#FFD966"), Array("Orange", "#ED7D31"), _
              Array("Brown", "#7F6000"))
    wsM.Columns("AB").ColumnWidth = 3
    wsM.Columns("AC:AD").ColumnWidth = 14

    Set lo = GetTable(SH_DB, TBL_DEFECTS)
    EnsureDefectColorColumn lo

    If StrComp(CStr(ws.Range("A8").Value), "Paint Color", vbTextCompare) <> 0 Then
        ws.Rows(8).RowHeight = 22
        ws.Range("A8").Value = "Paint Color"
        With ws.Range("A8")
            .Interior.Color = RGB(31, 78, 121)
            .Font.Color = RGB(255, 255, 255)
            .Font.Bold = True
            .VerticalAlignment = xlCenter
        End With
        With ws.Range("B8")
            .NumberFormat = "@"
            .Interior.Color = RGB(255, 242, 204)
            .Font.Bold = True
            .HorizontalAlignment = xlLeft
            .VerticalAlignment = xlCenter
        End With
        With ws.Range("C8:D8")
            .Merge
            .WrapText = True
            .VerticalAlignment = xlCenter
            .Font.Italic = True
            .Font.Size = 9
            .Font.Color = RGB(90, 90, 90)
        End With
        ws.Range("C8").Value = "Mau son xe (bat buoc). Danh sach: MASTER > tblColors."
        With ws.Range("A8:D8").Borders
            .LineStyle = xlContinuous
            .Color = RGB(150, 150, 150)
            .Weight = xlThin
        End With
        RepositionPanelButtons ws
    End If
    ws.Range("B8").Locked = False
    Exit Sub
EH:
    MsgBox "EnsureColorSupport loi: " & Err.Description, vbExclamation
End Sub

'Them cot PaintColor vao cuoi tblDefects neu chua co (du lieu cu de trong)
Private Sub EnsureDefectColorColumn(ByVal lo As ListObject)
    Dim lc As ListColumn, found As Boolean
    For Each lc In lo.ListColumns
        If StrComp(lc.Name, "PaintColor", vbTextCompare) = 0 Then
            found = True
            Exit For
        End If
    Next lc
    If Not found Then
        Set lc = lo.ListColumns.Add
        lc.Name = "PaintColor"
    End If
    If Not lo.DataBodyRange Is Nothing Then lo.ListColumns("PaintColor").DataBodyRange.NumberFormat = "@"
    lo.Parent.Columns(lo.ListColumns("PaintColor").Range.Column).ColumnWidth = 16
End Sub

'Hang 8 cao hon (8 -> 22 pt) nen cac nut o panel trai duoc dich xuong theo
Private Sub RepositionPanelButtons(ByVal ws As Worksheet)
    Dim arr As Variant, i As Long, s As Shape
    arr = Array(Array("BTN_SAVE", "A9:D11"), Array("BTN_CLEAR", "A12:B13"), Array("BTN_RESET", "C12:D13"), _
                Array("BTN_SHOW", "A27:B28"), Array("BTN_HIDE", "C27:D28"))
    For i = 0 To UBound(arr)
        Set s = FindShape(ws, CStr(arr(i)(0)))
        If Not s Is Nothing Then s.Top = ws.Range(CStr(arr(i)(1))).Top + 2
    Next i
End Sub
